def wrap_positions(pos, L, W):
    """Map positions in place to the periodic box ``[0, L) x [0, W)``."""
    pos[:, 0] = np.mod(pos[:, 0], L)
    pos[:, 1] = np.mod(pos[:, 1], W)
    return pos
fast_pairs.pyPeriodic candidate-pair searches with a k-d treeExcerpt

The accelerated search preserves the engine’s periodic pair convention.

def _kdtree_pairs_periodic(pos, L, W, r_query):
    """Return periodic pairs within ``r_query`` using the engine contract."""
    p = np.mod(np.asarray(pos, float), [L, W])
    # cKDTree with boxsize requires points strictly inside [0, L) x [0, W)
    p[p[:, 0] >= L, 0] = 0.0
    p[p[:, 1] >= W, 1] = 0.0
    tree = cKDTree(p, boxsize=[L, W])
    prs = tree.query_pairs(r=float(r_query), output_type="ndarray")
    if len(prs) == 0:
        return _EMPTY
    return (prs[:, 0].astype(np.int64), prs[:, 1].astype(np.int64))

dataset.pyDataset geometry, component ordering and wall-normal conventionsExcerpt

A local three-point Lagrange derivative respects the stored wall-normal node positions.

def _lagrange3_derivative(F3, y3, at=1):
    """d/d(distance) at node `at` from three nodes at true positions y3."""
    z0, z1, z2 = y3
    zc = y3[at]
    dL0 = ((zc - z1) + (zc - z2)) / ((z0 - z1) * (z0 - z2))
    dL1 = ((zc - z0) + (zc - z2)) / ((z1 - z0) * (z1 - z2))
    dL2 = ((zc - z0) + (zc - z1)) / ((z2 - z0) * (z2 - z1))
    return F3[:, :, 0] * dL0 + F3[:, :, 1] * dL1 + F3[:, :, 2] * dL2

run_study.pyMain command-line configuration and study workflowExcerpt

One resolver translates a model label into the contact parameters used downstream.

def model_of(cfg):
    """
    (model letter, effective e_tan, effective mu_pp) from cfg["model"].
    e_tan is 1 (smooth) unless the model is C, D or E; mu_pp is -1 (no cap)
    unless the model is D or E. An unknown letter is reported as itself so
    validate() can name it.
    """
    m = str(cfg.get("model", "A")).strip().upper()
    e_tan = float(cfg.get("e_tan", 0.0)) if m in ("C", "D", "E") else 1.0
    mu = float(cfg.get("mu_pp", 0.3)) if m in ("D", "E") else -1.0
    return m, e_tan, mu

def _asnap(v):
    """'all' / 'none' / '' -> None (no clamp); otherwise an int."""
    if v is None:
        return None
    if isinstance(v, str):
        if v.strip().lower() in ("all", "none", ""):
            return None
        v = int(v)
    v = int(v)
    # Negative values use the same "no clamp" convention as run_study. Convert
    # them here before the downstream frame-count clamp.
    return None if v < 0 else v

clustering.pyPeriodic clustering measures and finite-sample referencesExcerpt

The variance-to-mean ratio is paired with grid occupancy for context.

def vmr(p, L, W, nbins):
    """Variance-to-mean ratio and occupancy on an nbins x nbins grid."""
    H, _, _ = np.histogram2d(np.mod(p[:, 0], L), np.mod(p[:, 1], W),
                             bins=nbins, range=[[0, L], [0, W]])
    c = H.ravel()
    return float(c.var() / c.mean()), float(100.0 * np.mean(c > 0))
collective.pyPair motion, collision transport and orientational statisticsExcerpt

The contact-scale diagnostic separates mean relative speed from approaching motion.

def w_at_contact(p, v, L, W, sigma, width=1.5):
    """
    Mean |w_r| and mean approaching w_r^- for pairs within `width`*sigma.
    This is the transport factor of the collision kernel.
    """
    ij, d, r = _pairs(p, L, W, width * sigma)
    _ok = r > 0                      # coincident pairs: no direction
    ij, d, r = ij[_ok], d[_ok], r[_ok]
    if len(ij) == 0:
        return {"w_abs": np.nan, "w_minus": np.nan,
                "frac_approach": np.nan, "n_pairs": 0}
    rhat = d / r[:, None]
    wr = np.sum((v[ij[:, 0]] - v[ij[:, 1]]) * rhat, axis=1)
    return {"w_abs": float(np.abs(wr).mean()),
            "w_minus": float(-wr[wr < 0].mean()) if np.any(wr < 0) else 0.0,
            "frac_approach": float(np.mean(wr < 0)),
            "n_pairs": int(len(wr))}
structures.pyFlow gradients, streaks and particle–structure relationshipsExcerpt

Periodic directions use a spectral derivative on the DNS grid.

def d_spectral(F, axis, h):
    """Exact derivative of a periodic field along `axis` with spacing `h`."""
    n = F.shape[axis]
    k = np.fft.fftfreq(n, d=h / (2 * np.pi))
    sh = [1, 1]
    sh[axis] = n
    return np.real(np.fft.ifft(1j * k.reshape(sh) * np.fft.fft(F, axis=axis),
                               axis=axis))
vertical_impulse.pyPost-processing of diagnostic contact impulsesExcerpt

Archived slip components allow a diagnostic impulse to be re-evaluated for another contact law.

def pz_for_law(d, e_tan, mu_pp, m=1.0):
    """Re-evaluate Pz~ from the archived slips for another (e_tan, mu_pp)."""
    wt = d["collision_wt"].astype(float)
    uy = d["collision_Uy_diag"].astype(float)
    Pn = d["collision_Pn"].astype(float)
    uT = np.hypot(wt, uy)
    trial = (m / 7.0) * (1.0 - e_tan) * uT
    if mu_pp >= 0:
        trial = np.minimum(trial, mu_pp * Pn)
    with np.errstate(divide="ignore", invalid="ignore"):
        pz = np.where(uT > 0, -trial * uy / uT, 0.0)
    return pz

The collision kernel is displayed beside its accumulation and transport factors.

def fig_collision_kernel(rows, path=None):
    """
    The collision kernel split into its two factors:
        Gamma = n sigma * g(sigma) * <|w_r|>
                        accumulation   transport
    Plotting them separately says WHY the collision rate changed with Stokes
    number, which the total rate alone cannot.
    """
    st = np.array([r["tau"] for r in rows])
    g = np.array([r.get("g_contact", np.nan) for r in rows], float)
    w = np.array([r.get("w_abs_contact", np.nan) for r in rows], float)
    fig, ax = plt.subplots(1, 3, figsize=(fs.WIDE, 2.2))
    for a, y, lbl, col in ((ax[0], g, "$g(\\sigma)$  (accumulation)", BLUE),
                           (ax[1], w, "$\\langle|w_r|\\rangle$  (transport)", RED),
                           (ax[2], g * w, "$g\\cdot\\langle|w_r|\\rangle$  (kernel)", NAVY)):
        a.plot(st, y, "o-", color=col, lw=1.8)
        a.set_xscale("symlog", linthresh=0.05)
        a.set_xlabel(fs.SYMBOL["tau"]); a.set_ylabel(lbl)
    ax[0].axhline(1.0, color=GREY, ls="--", label="uniform")
    ax[0].legend()
    fs.panels(ax)
    fig.tight_layout()
    return _save(fig, path)
viz.pyFlow-field, particle-density and animation utilitiesExcerpt

A periodic roll removes the measured downstream translation from an animation.

def comoving_roll(F, t, U_c=U_CONV, L=LX, hx=HX):
    """
    Shift a field into a frame moving downstream at U_c.

    The streamwise axis is periodic, so this is a circular roll by the integer
    number of cells the frame has travelled. Sub-cell precision is not needed:
    the residual is under half a cell, far below the streak width.
    """
    n = int(round((U_c * t / hx))) % F.shape[1]
    return np.roll(F, -n, axis=1)
snapshots.pyCompact particle-snapshot saving, loading and plottingExcerpt

Run metadata is restored separately from the time-dependent arrays.

def load_snapshots(path):
    """Read back a saved run. Returns a dict with positions, times and meta."""
    z = np.load(path, allow_pickle=True)
    out = {k: z[k] for k in z.files if not k.startswith("meta_")}
    out["meta"] = {k[5:]: z[k].item() if z[k].ndim == 0 else z[k]
                   for k in z.files if k.startswith("meta_")}
    return out

def panel(ax, letter, **kw):
    """
    Add a panel label to the left-aligned title.

    Keeping the label in the title avoids covering data or competing with a
    legend inside the axes.
    """
    t = ax.get_title()
    ax.set_title("")
    kw.setdefault("fontsize", 9)
    ax.set_title(f"({letter}) {t}" if t else f"({letter})", loc="left", **kw)
    return ax
housestyle.pySemantic plotting styles and shared comparison limitsExcerpt

Paired fields use one robust colour scale so visual contrast remains comparable.

def paired_limits(*fields, symmetric=True, q=0.995):
    """Return one colour scale for the fields being compared.

    Shared limits keep visual differences between paired fields comparable.
    """
    import numpy as np
    allv = np.concatenate([np.asarray(F, float).ravel() for F in fields])
    allv = allv[np.isfinite(allv)]
    if not allv.size:
        return (0.0, 1.0)
    if symmetric:
        a = float(np.quantile(np.abs(allv), q))
        return (-a, a)
    return (float(np.quantile(allv, 1 - q)), float(np.quantile(allv, q)))
